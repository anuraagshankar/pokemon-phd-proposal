% ---------------------------------------------------------------------------
% Pokeball decorative rule: full-width rule with a medallion at the center.
% Red upper half, black lower half, white center circle -- single red accent.
% ---------------------------------------------------------------------------
\newcommand{\pokeball}{%
  \begin{center}
    \begin{tikzpicture}
      \def\R{0.38}
      \def\r{0.14}
      \def\halfgap{0.52}

      % Left rule segment
      \draw[line width=0.4pt, black]
        (-\textwidth/2 + 1pt, 0) -- (-\halfgap cm, 0);
      % Right rule segment
      \draw[line width=0.4pt, black]
        (\halfgap cm, 0) -- (\textwidth/2 - 1pt, 0);

      % Red upper semicircle
      \fill[pokeRed]  (-\R cm, 0) arc (180:0:\R cm) -- cycle;
      % Black lower semicircle
      \fill[black]    (-\R cm, 0) arc (-180:0:\R cm) -- cycle;
      % White dividing stripe
      \draw[white, line width=1.4pt] (-\R cm, 0) -- (\R cm, 0);
      % Outer ring
      \draw[black, line width=0.5pt] (0,0) circle (\R cm);
      % Center button
      \fill[white] (0,0) circle (\r cm);
      \draw[black, line width=0.5pt] (0,0) circle (\r cm);
    \end{tikzpicture}
  \end{center}%
}
